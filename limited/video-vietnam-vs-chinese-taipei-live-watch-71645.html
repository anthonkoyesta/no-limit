Default tag use hobe na.") return [] data = read_file_lines(fallback_file, show_log=False) if data: if show_log: print(f"📄 {fallback_file.name} fallback theke {len(data)} ti data load hoyeche.") return data if show_log: print(f"⚠️ Warning: {resource_type} er jonno category ba any file-e data nei.") return [] def clean_output_folder(output_folder): """Protibar generate korar age output folder er old file delete kore.""" output_folder = Path(output_folder) if output_folder.exists(): for item in output_folder.iterdir(): try: if item.is_file() or item.is_symlink(): item.unlink() elif item.is_dir(): shutil.rmtree(item) except Exception as e: print(f"⚠️ Warning: {item.name} delete kora jayni: {e}") print(f"\n🗑️ '{output_folder}' folder theke sob purano file delete kora hoyeche.") else: output_folder.mkdir(parents=True, exist_ok=True) def replace_placeholders(content, replacements): """Placeholder replace kore.""" for key, value in replacements.items(): pattern = r"{{\s*" + re.escape(key) + r"\s*}}" content = re.sub(pattern, lambda m: str(value), content) return content # ========================= # Choice Menus # ========================= def get_category_choice(): category_map = { "1": "nfl", "2": "ncaaf", "3": "ncaaw", "4": "soccer", "5": "highschool", "6": "rugby", "7": "any", "8": "movie", "9": "leak", } while True: print("\n" + "=" * 60) print(" 🎯 POST TYPE SELECT KORUN ".center(60, "=")) print("=" * 60) print(" 1. """ print("\n" + "=" * 70) print(" 🎲 RANDOM POST GENERATION MODE ".center(70, "=")) print("=" * 70) print(f"📌 Apnar event file e sudhu 1 ti event pawa geche:") print(f" → '{event_name}'") print("-" * 70) print("💡 Apni ki ei 1 ta event theke onek random post banate chan?") print(" (Prottek post e alada tag, image, filename part thakbe)") print("=" * 70) while True: choice = input("\n👉 H/N (H = Hya, N = Na): ").strip().lower() if choice in ["n", "na", "no"]: print("✅ Thik ache, sudhu 1 ta post generate hobe.") return 1 if choice in ["h", "ha", "yes", "y"]: while True: count_input = input("\n📊 Koto gulo random post banaben? [{url_type}] {display_url}{recommend_mark}") print("-" * 75) print(" Enter = recommended URL thakle auto select") print(" r = random URL") print("=" * 75) while True: choice = input("\n👉 URL choice din: ").strip().lower() if choice == "" and recommended_index is not None: url_type, selected_url = options[recommended_index - 1] print(f"✅ Recommended URL selected [{url_type}]: {selected_url}") return selected_url, url_type if choice == "r": url_type, selected_url = random.choice(options) print(f"✅ Random URL selected [{url_type}]: {selected_url}") return selected_url, url_type if choice.isdigit(): idx = int(choice) if 1 <= idx <= len(options): url_type, selected_url = options[idx - 1] print(f"✅ URL selected [{url_type}]: {selected_url}") return selected_url, url_type print("❌ Invalid choice! Number din, ba Enter/r use korun.") def get_leak_url(): """Leak category er jonno specific URL file theke URL read kore.""" leak_url_file = SCRIPT_DIR / "leak_url.txt" if not leak_url_file.exists(): leak_url_file = RESOURCE_DIR / "leak" / "leak_url.txt" urls = read_file_lines(leak_url_file, show_log=False) if urls: selected_url = urls[0] print(f"\n✅ Leak URL auto-selected: {selected_url}") return selected_url, "LEAK URL" else: print("\n❌ Error: 'leak_url.txt' file pawa jayni ba khali!") print(" Doya kore main folder ba 'resource_files/leak/' folder e 'leak_url.txt' create korun.") return None, None def ask_continue(): print("\n" + "=" * 60) while True: choice = input("🔄 Abar notun kore generate korben? (Y/N): ").strip().lower() if choice in ["y", "yes"]: return True if choice in ["n", "no"]: return False print("❌ Invalid input! Y ba N din.") # ========================= # Main Generator # ========================= def generate_html_files(category, template_path, selected_url, selected_url_type, custom_text, output_folder, posts_per_event=1): clean_output_folder(output_folder) #
